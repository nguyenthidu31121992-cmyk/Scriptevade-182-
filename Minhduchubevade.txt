local function isSpawnedNextbot(obj: Instance): boolean
	if not obj:IsA("Model") then return false end
	if Players:GetPlayerFromCharacter(obj) then return false end
	if obj == workspace or obj.Parent == CoreGui then return false end

	local humanoid = obj:FindFirstChildOfClass("Humanoid")
	local hasAnimator = obj:FindFirstChild("ClientAnimator", true) or obj:FindFirstChild("AnimationController", true)
	
	if humanoid or hasAnimator then
		local primary = obj.PrimaryPart or obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildOfClass("Part")
		if primary then
			if primary.Transparency < 1 then
				return true
			end
		end
	end

	return false
end

local function isValidBubble(obj: Instance): boolean
	if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("TextLabel") then return false end
	
	local nameLower = obj.Name:lower()
	if nameLower == "bubble" or nameLower:find("bubble") or nameLower:find("bong bóng") then
		if obj:IsA("Model") or obj:IsA("BasePart") then
			local part = obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildOfClass("Part")) or (obj:IsA("BasePart") and obj)
			if part and part.Transparency < 1 then
				return true
			end
		end
	end
	return false
end

local function createFloatingButton(name: string, text: string, defaultPos: UDim2, getState, setState, onToggleChanged)
	local btn = Instance.new("TextButton")
	btn.Name = name
	btn.Size = UDim2.new(0, 125, 0, 42)
	btn.Position = defaultPos
	btn.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
	btn.BackgroundTransparency = 0.2
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.TextSize = 11
	btn.FontFace = Font.new("rbxasset://fonts/families/GothamBold.json")
	btn.Text = text
	btn.Visible = true
	btn.Parent = screenGui
	Instance.new("UICorner", btn).CornerRadius = UDim.new(1, 0)
	local stroke = Instance.new("UIStroke", btn)
	stroke.Thickness = 2

	local activeInput: InputObject? = nil
	local dragStartPos = Vector2.zero
	local startPosBtn = UDim2.new()
	local hasMoved = false

	btn.InputBegan:Connect(function(input)
		if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) and not activeInput then
			activeInput = input
			dragStartPos = input.Position
			startPosBtn = btn.Position
			hasMoved = false
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == activeInput then
			local delta = input.Position - dragStartPos
			if delta.Magnitude > 6 then
				hasMoved = true
			end
			btn.Position = UDim2.new(startPosBtn.X.Scale, startPosBtn.X.Offset + delta.X, startPosBtn.Y.Scale, startPosBtn.Y.Offset + delta.Y)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input == activeInput then
			if not hasMoved then
				local newState = not getState()
				setState(newState)
				btn.BackgroundColor3 = newState and Color3.fromRGB(0, 255, 100) or Color3.fromRGB(20, 20, 25)
				if onToggleChanged then
					onToggleChanged(newState)
				end
			end
			activeInput = nil
		end
	end)

	return btn, stroke
end

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then draggingSlider = true end
